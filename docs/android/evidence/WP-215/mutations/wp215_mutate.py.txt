def failing_tests():
    names = []
    for path in glob.glob(WT + "/android/core/services/build/test-results/test/TEST-*.xml"):
        for case in ET.parse(path).getroot().iter("testcase"):
            if any(child.tag in ("failure", "error") for child in case):
                names.append(case.get("name"))
    return names


def run(label, filename, old, new):
    path = PKG + filename
    backup = SCRATCH + "wp215_backup_" + filename
    shutil.copyfile(path, backup)
    try:
        source = open(path, encoding="utf-8").read()
        if source.count(old) != 1:
            return f"{label}: SKIPPED (pattern count {source.count(old)})"
        open(path, "w", encoding="utf-8").write(source.replace(old, new))
        result = subprocess.run(
            ["timeout", "1200", "./gradlew", ":core:services:test", "--rerun", "--console=plain", "-q"],
            cwd=WT + "/android", env=ENV, capture_output=True, text=True,
        )
        failed = failing_tests()
        verdict = "KILLED" if result.returncode != 0 and failed else "SURVIVED"
        return f"{label}: {verdict} ({len(failed)} failing) {failed}"
    finally:
        shutil.copyfile(backup, path)
        os.remove(backup)


if __name__ == "__main__":
    selected = {"review": REVIEW_MUTATIONS, "room": MUTATIONS[:1]}.get(" ".join(sys.argv[1:]), MUTATIONS + REVIEW_MUTATIONS)
    for mutation in selected:
        print(run(*mutation), flush=True)
    sys.exit(0)

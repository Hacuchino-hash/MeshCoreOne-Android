// WP-317 parity oracle 2: grouping/exponent edge cases of the lenient TextField number parsing.
import Foundation
func show(_ label: String, _ value: Any?) { print("\(label)\t\(value.map { "\($0)" } ?? "nil")") }
let posix = Locale(identifier: "en_US_POSIX")
let freqStyle = FloatingPointFormatStyle<Double>.number.precision(.fractionLength(3)).locale(posix)
let intStyle = IntegerFormatStyle<Int>.number.locale(Locale(identifier: "en_US"))
print("== Int en_US")
for s in ["12,34", "1,00", "1,0000", "1234,567", "1,234,567", ",5", "1,000.5", "1.5e1", "1e", "1e+2", "1E2", "-1e1", "- 5", "+ 5", "--5", "5-", "1 000", "0", "-0", "+0", "00012", "9223372036854775807", "9223372036854775808", "-9223372036854775808", "1e30", "  ", "\t7", "7\n", "1.e1", ".5", "-.5", "1.9", "-1.9"] {
  show("[\(s)]", try? intStyle.parseStrategy.parse(s))
}
print("== Double posix")
for s in ["1e", "1e+2", "1E2", "1e-2", "-1e1", "- 5", "--5", "5-", "\t915", "915\n", "1.5e1", "1.e1", "-.5", "+.5", "-0", "NaN", "Infinity", "-inf", "1e400", "1e-400", "0.0005", "915.00049999", "  ", "9,15", "1 000", "00915.0"] {
  show("[\(s)]", try? freqStyle.parseStrategy.parse(s))
}

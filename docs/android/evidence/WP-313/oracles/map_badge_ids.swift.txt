func f(_ d: Double, _ id: String) -> String {
  Measurement(value: d, unit: UnitLength.meters).formatted(.measurement(width: .abbreviated, usage: .road).locale(Locale(identifier: id)))
}

}
// Stable ids
func stableID(_ role: UInt8, _ key: Data) -> UUID {
  var m = Data([role]); m.append(key)
  let b = Array(SHA256.hash(data: m))
  return UUID(uuid: (b[0],b[1],b[2],b[3],b[4],b[5],b[6],b[7],b[8],b[9],b[10],b[11],b[12],b[13],b[14],b[15]))
}

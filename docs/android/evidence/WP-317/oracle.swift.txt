func show(_ label: String, _ value: Any?) {
  print("\(label)\t\(value.map { "\($0)" } ?? "nil")")
}

print("== formatBandwidth fallback (NumberFormatter en_US_POSIX, 0...2 fraction digits)")
func fmtBW(_ hz: UInt32) -> String {
  let khz = Double(hz) / 1000.0
  if khz.truncatingRemainder(dividingBy: 1) == 0 { return "\(Int(khz))" }
  let f = NumberFormatter()
  f.locale = Locale(identifier: "en_US_POSIX")
  f.minimumFractionDigits = 0
  f.maximumFractionDigits = 2
  return f.string(from: NSNumber(value: khz)) ?? "\(khz)"
}

print("== nearestBandwidth")
let bws: [UInt32] = [7800, 10400, 15600, 20800, 31250, 41700, 62500, 125_000, 250_000, 500_000]
func nearest(_ hz: UInt32) -> UInt32 {
  if bws.contains(hz) { return hz }
  return bws.min { l, r in
    let ld = l > hz ? l - hz : hz - l
    let rd = r > hz ? r - hz : hz - r
    return ld < rd
  } ?? 250_000
}
for hz in [7799, 9100, 9101, 13000, 18200, 26000, 36475, 52100, 93750, 187500, 375000, 0, 900000, 4294967295] as [UInt32] { show("\(hz)", nearest(hz)) }

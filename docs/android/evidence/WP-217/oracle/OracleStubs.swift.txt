public enum OracleClock {

}

public enum ProtocolLimits { public static let publicKeySize = 32 }
public enum PathEncoding { public static let maxHopCount = 63 }

public func encodePathLen(hashSize: Int, hopCount: Int) -> UInt8 {
  precondition(1...3 ~= hashSize, "hashSize must be 1, 2, or 3")
  let mode = UInt8(hashSize - 1)
  let hops = UInt8(min(hopCount, PathEncoding.maxHopCount))
  return (mode << 6) | hops
}

public enum MessageStatus: Int { case pending = 0, sending, sent, delivered, failed, retrying }
public enum MessageDirection: Int { case incoming = 0, outgoing }
public enum TextType: UInt8 { case plain = 0, cliData, signedPlain }
public enum ContactType: UInt8 { case chat = 1, repeater, room }
public enum NotificationLevel: Int { case muted = 0, mentionsOnly, all }
public enum DecryptStatus: Int {
  case notApplicable = 0, noMatchingKey, hmacFailed, decryptFailed, success, pending, dmNoMatchingKey
}
public enum RouteType: UInt8 { case tcFlood = 0, flood, direct, tcDirect }
public enum PayloadType: UInt8 {
  case request = 0, response, textMessage, ack, advert, groupText, groupData, anonRequest, path, trace, multipart
}

public struct ParsedRxLogData {
  public let snr: Double?, rssi: Int?, rawPayload: Data, routeType: RouteType, payloadType: PayloadType
  public let payloadVersion: UInt8, payloadTypeBits: UInt8, transportCode: Data?, pathLength: UInt8
  public let pathNodes: [UInt8], packetPayload: Data, packetHash: String

  public init(
    snr: Double?, rssi: Int?, rawPayload: Data, routeType: RouteType, payloadType: PayloadType,
    payloadVersion: UInt8, payloadTypeBits: UInt8, transportCode: Data?, pathLength: UInt8,
    pathNodes: [UInt8], packetPayload: Data
  ) {
    self.snr = snr; self.rssi = rssi; self.rawPayload = rawPayload; self.routeType = routeType
    self.payloadType = payloadType; self.payloadVersion = payloadVersion; self.payloadTypeBits = payloadTypeBits
    self.transportCode = transportCode; self.pathLength = pathLength; self.pathNodes = pathNodes
    self.packetPayload = packetPayload
    // MeshCore ParsedRxLogData.computePacketHash(from:)
    packetHash = SHA256.hash(data: packetPayload).prefix(8).map { String(format: "%02x", $0) }.joined()
  }
}
